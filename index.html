  <script>

    const form = document.getElementById('form');

    const out = document.getElementById('out');


    function getCheckedValues(name) {

      return [...form.querySelectorAll(`input[name="${name}"]:checked`)].map(x => x.value);

    }

    function getSingleValue(name) {

      const el = form.querySelector(`input[name="${name}"]:checked`);

      return el ? el.value : "(not picked)";

    }

    function getText(name) {

      const el = form.querySelector(`[name="${name}"]`);

      return el && el.value.trim() ? el.value.trim() : "(blank)";

    }


    function buildMessage() {

      const chocTypes = getCheckedValues("chocType");

      const msg =

`🎁 Birthday Picks


1) Gift vibe: ${getSingleValue("gift")}

2) Cake: ${getSingleValue("cake")}

3) Chocolate: ${chocTypes.length ? chocTypes.join(", ") : "(not picked)"} | Nuts: ${getSingleValue("nuts")}

4) Flowers/Plant: ${getSingleValue("green")}

5) What would make me happiest: ${getText("note")}

6) Delivery: City = ${getText("city")} | When = ${getText("timing")}


💛 Replying with love: I accept the virtual hugs + kisses now, and I’m claiming the real ones in summer.`;


      return msg;

    }


    document.getElementById('makeMsg').addEventListener('click', () => {

      out.style.display = "block";

      out.textContent = buildMessage();

      out.scrollIntoView({ behavior: "smooth", block: "start" });

    });


    document.getElementById('copyBtn').addEventListener('click', async () => {

      const text = out.textContent || buildMessage();

      try {

        await navigator.clipboard.writeText(text);

        alert("Copied! Now paste it into your chat ✨");

      } catch (e) {

        alert("Couldn’t auto-copy on this browser. You can manually select + copy the text.");

      }

    });

  </script>
